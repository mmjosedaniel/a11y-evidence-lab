$m604Package = Get-Content package.json -Raw | ConvertFrom-Json -AsHashtable

$m604Count = 0
foreach ($m604Group in @('dependencies','devDependencies')) {
  $m604Declared = $m604Package[$m604Group]
  $m604Locked = $m604Lock.packages[''][$m604Group]
  if ($m604Declared.Count -ne $m604Locked.Count) { throw 'Dependency count mismatch' }
  foreach ($m604Name in $m604Declared.Keys) {
    $m604Pin = $m604Declared[$m604Name]

        $m604Locked[$m604Name] -cne $m604Pin -or
        $m604Lock.packages['node_modules/' + $m604Name].version -cne $m604Pin) {
      throw 'Dependency pin mismatch'
    }
    $m604Count++
  }
}
